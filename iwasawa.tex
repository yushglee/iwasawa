\section{L-functions}


Throughout the note, 
$\F$ is a totally real number field of degree $d=[\F:\Q]$ 
and $\K$ is a totally imaginary quadratic extension of $\F$.
We let $\Delta_\F$
and $\Delta_\K$ denote the absolute discriminant of $\F$ and $\K$ respectively.

\[
	\begin{tikzcd}[column sep=tiny,row sep=tiny]
        &  & H_\infty \\
	\F_\infty \arrow[rru,dash] & & \\
        &  & \\
        &  & H=\F_{\psi}(\mu_p) \arrow[uuu,dash] \\
	\F \arrow[uuu,dash] \arrow[rru,dash] & &                     
	\end{tikzcd}
\]

\subsection{Class number formula}

Let $E^+=\oo_\F^\times$ 
and $E=\oo_\K^\times$
denote the multiplicative groups of units in $\F$ and $\K$ respectively,
and let $W\subset E$ denote the subgroups of roots of unity in $\K$.
By Dirichlet's unit theorem,
both groups have the same ranks $r=d-1$.
And furthermore, the number $Q\coloneqq [E:WE^+]$ 
is less than or equal to $2$ by \cite[Thm 4.12]{Washington},
where the equality holds if and only if $W=E^{c-1}$.

Now, the class number formulae for $\K$ and $\F$ 
and that $R_\K/R_\F=2^r Q^{-1}$ (\cite[Thm 4.16]{Washington})
imply that
\[
	\left.\frac{\zeta_\K(s)}{\zeta_\F(s)}\right\vert_{s=1}=
	\frac{(2\pi)^{d}h^-(\K)}{Qw\sqrt{\Delta_\K/\Delta_\F}}.
\]

When $\K/\Q$ is abelian, 
the left-hand side can be decomposed into L-values
of odd Dirichlet characters.
Combined with the conductor-discriminant formula
$\sqrt{-1}^d\cdot \sqrt{\Delta_\K/\Delta_\F}
=\prod_{\chi \text{ odd}}\mathfrak{g}(\chi)$,
we obtain the formula
\[
	h^-=Qw\prod_{\chi \text{ odd}}(-\frac{1}{2}B_{1,\chi})
	=Qw\prod_{\chi \text{ odd}}\frac{L(0,\chi)}{2}.
\]

In the particular case when 
$\K=\Q(\zeta_{p^{n+1}N})$, where $p\nmid N$,
we will further decompose the above into
\[
	h^-
	=Qw\prod_{\xi\in \mu_{p^n}}(-\frac{1}{2}B_{1,\omega^{-1}\psi_\zeta})
	\prod_{\substack{\chi \text{ odd}\\\chi\not\equiv \omega^{-1}\bmod p}}
	(-\frac{1}{2}B_{1,\chi})
\]
where the product of the first three terms is a $p$-unit.
Indeed, the Kubota--Leopoldt $p$-adic L-function
satisfies the interpolation formula
\[
	\phi_{k,\xi}(G_{\psi}H_{\psi})=
	(1-\psi \psi_\xi \omega^{-k}(p))
	L(1-k,\psi \psi_\xi \omega^{-k})\quad
	\text{ for }k\geq 1.
\]
And we have $G_\psi\in \Lambda^\times, H_\psi(T)=T$
when $\psi$ is the trivial character.
Therefore, the product of
\[
	\prod_{\xi\in \mu_{p^n}}(-\frac{1}{2}B_{1,\omega^{-1}\psi_\zeta})
	\sim \prod_{\xi\in \mu_{p^n}}L(0,\psi_\xi \omega^{-1})
	\sim \prod_{\xi\in \mu_{p^n}}\phi_{1,\xi}(T)^{-1}
	=\prod_{\xi\in \mu_{p^n}}(\xi u-1)^{-1}
	=(u^{p^n}-1)^{-1},
\]
where $u\in \Zp^\times$ is a topological generator of $1+p\Zp$,
with $Qw\sim p^{n+1}$ is a $p$-unit.
Here we write $a\sim b$ if the quotient $b/a$ of two numbers
$a$ and $b$ is a $p$-unit.

\begin{thm}[Herbrand--Ribet]
	Let $A=Cl_{\Q(\mu_p)}\otimes\Zp=\bigoplus_{i=0}^{p-2}A_{i}$
	be the decomposition 
	induced by the action of $Gal(\Q(\mu_p)/\Q)$
	on the ideal class group,
	where $A_i$ is the component on which 
	the Galois group acts by $\omega^i$.
	Then when $3\leq i\leq p-2$ is an odd number,
	$A^i\neq 0$ if and only if  
	$p\mid B_{p-i}\sim L(0,\omega^{-i})$.
\end{thm}

\begin{rem}
	It is known that both $A_0$ and $A_1$ are always trivial.
	This agrees with the class number formula
\[
	h^-=p\prod_{i \text{ odd}}(-\frac{1}{2}B_{1,\omega^{i}})
	\sim pB_{1,\omega^{-1}}\cdot
	\prod_{\substack{i=3 \\i\text{ odd}}}^{p-2}(B_{1,\omega^{i}})
\]
since $pB_{1,\omega^{-1}}$ is a $p$-unit.
On the other hand, the Vandiver conjectures expects that 
$A_i=0$ for all $i$ even.
\end{rem}

\begin{rem}
	We can describe (the dual of) the component $A_i$
	through the Selmer group
	\[
	H^1_f(\Q, \mathbb{F}(\omega^i))
	=\{\phi\colon Cl_{\Q(\mu_p)}\to \mathbb{F}\mid
	\phi(x)=\omega^i(\gamma)\cdot \phi(\gamma^{-1}x \gamma)\}
	\cong \Hom(A_i,\mathbb{F}).
	\]
	It may seem strange that the Selmer group for $\omega^i$
	is related to the L-value of $\omega^{-1}$.
	This is due to the fact that we have identified
	Dirichlet characters with characters of Galois groups
	through the arithmetic normalization,
	but the Bloch--Kato conjecture
	relates Selmer groups to the L-values
	defined by the Euler products of characteristic polynomials
	of the geometric Frobenii.
\end{rem}


\subsection{Main conjecture over $\Q$}

Let $\psi$ be an even Dirichlet character.
We dnote $\Psi=\psi\langle*\rangle$
and $\Psi^*=\epsilon \Psi^{-1}$.
We then define
\[
	H_f^1(\Q,\Psi^*)=\ker
	\bigg\{
		H^1(\Q^\Sigma/\Q,\Psi^*)\to
		\prod_{v\in \Sigma}H^1(I_v,\Psi^*)
	\bigg\}
\]
when $\Sigma$ contains $p$ and all the places where
$\psi$ is ramified. 
Moreover, if $\psi$ is ramified at a place $\ell\neq p$
and let $1\to I''\to I_\ell\to I'\to 1$,
where $I'\cong \Zp(1)$
and $I''$ is prime-to-$p$, then
\[
	H^1(I_\ell,\Psi^*)\cong H^1(I',(\Psi^*)^{I''})\cong
	(\Psi^*)^{I''}/(\Psi^*(I')-1)=
	(\Psi^*)^{I''}/(\psi^{-1}(I')-1)=0
\]
since either $\psi\vert_{I''}\neq 1$ and $(\Psi^*)^{I''}=0$,
or $\psi\vert_{I''}= 1$, thus
$\psi\vert_{I'}\neq 1$, and
$(\Psi^*)^{I''}/(\psi^{-1}(I')-1)=0$.
Therefore when $\Sigma$ contains exactly $p$ 
and the primes where $\psi$ is ramified, then 
\[
	H_f^1(\Q,\Psi^*)=\ker
	\bigg\{
		H^1(\Q^\Sigma/\Q,\Psi^*)\to
		H^1(I_p,\Psi^*)
	\bigg\}
\]
